CREATE TABLE BOOKS (
	ID INT PRIMARY KEY IDENTITY (1, 1),
	NAME VARCHAR(50) NOT NULL,
	PRICE INT NOT NULL,
	CATEGORYID INT NOT NULL,
	AUTHORID INT NOT NULL;
);

CREATE TABLE CATEGORIES (
ID INT NOT NULL AUTO_INCREMET PRIMARY KEY, 
NAME VARCHAR(50) NOT NULL
);

CREATE TABLE AUTHORS (
ID INT PRIMARY KEY, 
NAME VARCHAR(50) NOT NULL
);
